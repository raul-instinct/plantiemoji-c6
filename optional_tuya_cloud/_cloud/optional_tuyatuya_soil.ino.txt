#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>      // Library Manager: ArduinoJson
#include "mbedtls/md.h"
#include "mbedtls/sha256.h"

const char *WIFI_SSID = "...", *WIFI_PASS = "...";
const char *TUYA_HOST = "https://openapi.tuyaeu.com";   // pick your project's data center (eu / us / in / cn)
const char *ACCESS_ID = "...", *ACCESS_SECRET = "...", *DEVICE_ID = "...";
const char *DP_CODE   = "humidity";                      // soil sensors usually report moisture as "humidity"; check the status list

String hmacHex(const String &key, const String &msg) {
  uint8_t out[32];
  mbedtls_md_hmac(mbedtls_md_info_from_type(MBEDTLS_MD_SHA256), (const uint8_t*)key.c_str(), key.length(), (const uint8_t*)msg.c_str(), msg.length(), out);
  String s; char b[3];
  for (int i = 0; i < 32; i++) { sprintf(b, "%02X", out[i]); s += b; }
  return s;                                              // Tuya wants UPPERCASE hex
}
String tuyaToken;
String tuyaTimeMs() { time_t n = time(nullptr); return String((uint64_t)n * 1000ULL); } // call configTime(0,0,"pool.ntp.org") after WiFi connects

String tuyaGet(const String &path, bool tokenCall) {
  String t = tuyaTimeMs();
  const String emptySha = "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855";
  String strToSign = "GET\n" + emptySha + "\n\n" + path;
  String sign = hmacHex(ACCESS_SECRET, String(ACCESS_ID) + (tokenCall ? "" : tuyaToken) + t + strToSign);
  WiFiClientSecure c; c.setInsecure();                   // use a pinned CA for anything permanent
  HTTPClient http; http.begin(c, String(TUYA_HOST) + path);
  http.addHeader("client_id", ACCESS_ID); http.addHeader("t", t); http.addHeader("sign", sign);
  http.addHeader("sign_method", "HMAC-SHA256");
  if (!tokenCall) http.addHeader("access_token", tuyaToken);
  int code = http.GET();
  String body = code > 0 ? http.getString() : "";
  http.end();
  return body;
}
bool tuyaGetToken() {
  JsonDocument d; deserializeJson(d, tuyaGet("/v1.0/token?grant_type=1", true));
  if (!d["success"]) return false;
  tuyaToken = d["result"]["access_token"].as<String>(); return true;   // valid ~2h; refresh when a call returns success:false
}
int tuyaSoilPercent() {                                  // -1 on failure
  if (tuyaToken.isEmpty() && !tuyaGetToken()) return -1;
  JsonDocument d; deserializeJson(d, tuyaGet(String("/v1.0/devices/") + DEVICE_ID + "/status", false));
  if (!d["success"]) { tuyaToken = ""; return -1; }
  for (JsonObject o : d["result"].as<JsonArray>()) if (String(o["code"].as<const char*>()) == DP_CODE) return o["value"].as<int>();
  return -1;
}
